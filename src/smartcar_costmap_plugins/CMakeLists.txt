cmake_minimum_required(VERSION 3.8)
project(smartcar_costmap_plugins)
find_package(ament_cmake REQUIRED)
find_package(nav2_costmap_2d REQUIRED)
find_package(pluginlib REQUIRED)
add_library(unknown_cost_layer SHARED src/unknown_cost_layer.cpp)
target_compile_features(unknown_cost_layer PUBLIC cxx_std_17)
target_include_directories(unknown_cost_layer PUBLIC
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
  $<INSTALL_INTERFACE:include>)
ament_target_dependencies(unknown_cost_layer nav2_costmap_2d pluginlib)
pluginlib_export_plugin_description_file(nav2_costmap_2d plugins.xml)
install(TARGETS unknown_cost_layer ARCHIVE DESTINATION lib LIBRARY DESTINATION lib)
install(DIRECTORY include/ DESTINATION include)
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(test_unknown_cost test/test_unknown_cost.cpp)
  target_link_libraries(test_unknown_cost unknown_cost_layer)
  ament_target_dependencies(test_unknown_cost nav2_costmap_2d)
endif()
ament_export_include_directories(include)
ament_export_libraries(unknown_cost_layer)
ament_export_dependencies(nav2_costmap_2d pluginlib)
ament_package()
